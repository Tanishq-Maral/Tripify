def IQR_based_outlier(series) :
    q1 = series.quantile(0.25)
    q3 = series.quantile(0.75)
    iqr = q3 - q1
    lb = q1 - 1.5 * iqr
    ub = q3 + 1.5 * iqr
    return lb, ub

X = df[['Age', 'EstimatedSalary']].values
# Target variable
y = df['Purchased'].values
# Step 4: Split Dataset
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.25,
    random_state=0
)

print(X_train.shape)
print("\nTesting Data Shape:")
print(X_test.shape)

sc = StandardScaler()

X_test_scaled = sc.transform(X_test)

log_reg = LogisticRegression(random_state=0)
# Train model
log_reg.fit(X_train_scaled, y_train)
# Model parameters
print("\nIntercept:")
print(log_reg.intercept_)
print("\nCoefficients:")
print(log_reg.coef_)
# Step 7: Predictions
# Predict class labels
y_pred = log_reg.predict(X_test_scaled)
# Predict probabilities
y_prob = log_reg.predict_proba(X_test_scaled)[:, 1]
print("\nFirst 10 Actual Values:")
print(y_test[:10])

print(y_pred[:10])
print("\nFirst 10 Predicted Probabilities:")

cm = confusion_matrix(y_test, y_pred)
print("\nConfusion Matrix:\n")
print(cm)

TN, FP, FN, TP = cm.ravel()
print("\nTrue Positives (TP):", TP)
print("False Positives (FP):", FP)

# Step 9: Evaluation Metrics
total = TP + TN + FP + FN
accuracy = (TP + TN) / total
error_rate = (FP + FN) / total
precision = TP / (TP + FP) if (TP + FP) != 0 else 0
recall = TP / (TP + FN) if (TP + FN) != 0 else 0
print("\nEvaluation Metrics:\n")
print("Accuracy:", round(accuracy, 3))
print("Error Rate:", round(error_rate, 3))
print("Precision:", round(precision, 3))

from sklearn.model_selection import train_test_split

from sklearn.metrics import (
    confusion_matrix,
    accuracy_score,
    classification_report
)
# Step 2: Load Dataset
# Make sure iris.csv is in the same folder as this Python file
iris = pd.read_csv("Datasets/assignment_6_iris_dataset.csv")
# Display first 5 rows
print("First 5 Rows:\n")
print(iris.head())
# Dataset information
print("\nDataset Info:\n")
print(iris.info())
# Count of each species
print("\nClass Distribution:\n")
print(iris['Species'].value_counts())
# Step 3: Prepare Features and Target
# Feature matrix (X)
X = iris[['SepalLengthCm',
          'SepalWidthCm',
          'PetalLengthCm',
          'PetalWidthCm']].values
# Target vector (y)
y = iris['Species'].values
# Step 4: Split Dataset into Train and Test
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.3,
    random_state=0,
    stratify=y
)
print("\nTraining Data Shape:", X_train.shape)
print("Testing Data Shape:", X_test.shape)
# Step 5: Create and Train Gaussian Naive Bayes Model
gnb = GaussianNB()
# Train the model
gnb.fit(X_train, y_train)
# Step 6: Predict on Test Data
y_pred = gnb.predict(X_test)
print("\nFirst 10 Actual Labels:")
print(y_test[:10])
print("\nFirst 10 Predicted Labels:")
print(y_pred[:10])
# Step 7: Confusion Matrix
labels = ['Iris-setosa', 'Iris-versicolor', 'Iris-virginica']
cm = confusion_matrix(y_test, y_pred, labels=labels)
print("\nConfusion Matrix:")
print(cm)
# Step 8: Accuracy
accuracy = accuracy_score(y_test, y_pred)
print("\nOverall Accuracy:", round(accuracy, 3))
# Step 9: Classification Report
print("\nClassification Report:\n")
print(classification_report(y_test, y_pred))
# Step 10: Calculate TP, FP, TN, FN and Metrics for Each Class
print("\n--- Metrics for Each Class ---\n")
for i in range(len(labels)):
    TP = cm[i, i]
    FP = cm[:, i].sum() - TP
    FN = cm[i, :].sum() - TP
    TN = cm.sum() - (TP + FP + FN)
    total = TP + FP + FN + TN
    accuracy_class = (TP + TN) / total
    error_rate = (FP + FN) / total
    precision = TP / (TP + FP) if (TP + FP) != 0 else 0
    recall = TP / (TP + FN) if (TP + FN) != 0 else 0
    print("Class:", labels[i])
    print("TP =", TP)
    print("FP =", FP)
    print("FN =", FN)
    print("TN =", TN)
    print("Accuracy   =", round(accuracy_class, 3))
    print("Error Rate =", round(error_rate, 3))
    print("Precision  =", round(precision, 3))
    print("Recall     =", round(recall, 3))

# Part 1: Text Preprocessing
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk import pos_tag
from nltk.stem import PorterStemmer, WordNetLemmatizer
from nltk.corpus import wordnet
# Download required NLTK resources (Run Once)
nltk.download('punkt')
nltk.download('punkt_tab')   # Required for newer NLTK versions
nltk.download('averaged_perceptron_tagger')
nltk.download('averaged_perceptron_tagger_eng')  # For newer versions
nltk.download('stopwords')
nltk.download('wordnet')
nltk.download('omw-1.4')
# 1. Sample Text
text = """
Text analytics is the process of turning unstructured text into meaningful data.
It includes steps like tokenization, stopword removal, stemming, and lemmatization.
"""
print("Original Text")
print(text)
# 2. Tokenization
tokens = word_tokenize(text)
print("Tokens")
print(tokens)
# 3. POS Tagging
pos_tags = pos_tag(tokens)
print("POS Tags")
print(pos_tags)
# 4. Stop-word Removal
stop_words = set(stopwords.words('english'))
tokens_no_sw = [
    word for word in tokens
    if word.lower() not in stop_words and word.isalpha()
]
print("Tokens after Stop-word Removal")
print(tokens_no_sw)
# 5. Stemming
stemmer = PorterStemmer()
stemmed_words = [stemmer.stem(word) for word in tokens_no_sw]
print("Stemmed Words")
print(stemmed_words)
# 6. Lemmatization
# Function to convert POS tags
def get_wordnet_pos(treebank_tag):
    if treebank_tag.startswith('J'):
        return wordnet.ADJ
    elif treebank_tag.startswith('V'):
        return wordnet.VERB
    elif treebank_tag.startswith('N'):
        return wordnet.NOUN
    elif treebank_tag.startswith('R'):
        return wordnet.ADV
    else:
        return wordnet.NOUN
lemmatizer = WordNetLemmatizer()
# POS tagging after stop-word removal
filtered_pos_tags = pos_tag(tokens_no_sw)
lemmatized_words = []
for word, tag in filtered_pos_tags:
    wn_tag = get_wordnet_pos(tag)
    lemma = lemmatizer.lemmatize(word.lower(), pos=wn_tag)
    lemmatized_words.append(lemma)
print("Lemmatized Words")
print(lemmatized_words)
# Part 2: TF-IDF Representation
from sklearn.feature_extraction.text import TfidfVectorizer
# Sample Corpus
corpus = [
    "Text analytics converts text into useful information.",
    "We apply tokenization, stopword removal, and lemmatization on documents.",
    "TF IDF gives higher weight to important words in a document."
]
print("Corpus")
for i, doc in enumerate(corpus, start=1):
    print(f"\nDocument {i}:")
    print(doc)
# Create TF-IDF Vectorizer
vectorizer = TfidfVectorizer()
# Fit and Transform
tfidf_matrix = vectorizer.fit_transform(corpus)
# Vocabulary Terms
terms = vectorizer.get_feature_names_out()
print("Vocabulary Terms")
print(terms)
# TF-IDF Matrix
print("TF-IDF Matrix")
print(tfidf_matrix.toarray())
# Non-zero TF-IDF Values for Document 1
doc1 = tfidf_matrix[0].toarray().flatten()
print("Non-zero TF-IDF values for Document 1")
for term, value in zip(terms, doc1):
    if value > 0:
        print(f"{term:15s} -> {value:.3f}")

import java.io.IOException;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.Mapper;
public class Mapper1 extends Mapper<Object, Text, Text, IntWritable> {
    public void map(Object key, Text value, Context context)
            throws IOException, InterruptedException {
        String line = value.toString();
        if (line.contains("ERROR"))
            context.write(new Text("ERROR"), new IntWritable(1));
        else if (line.contains("INFO"))
            context.write(new Text("INFO"), new IntWritable(1));
        else if (line.contains("WARN"))
            context.write(new Text("WARN"), new IntWritable(1));
    }
}
Step 4: Create Reducer Class
File: Reducer1.java
import java.io.IOException;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.Reducer;
public class Reducer1 extends Reducer<Text, IntWritable, Text, IntWritable> {
    public void reduce(Text key, Iterable<IntWritable> values, Context context)
            throws IOException, InterruptedException {
        int sum = 0;
        for (IntWritable v : values) {
            sum += v.get();
        }
        context.write(key, new IntWritable(sum));
    }
}

File: Driver.java
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.Job;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;
public class Driver {
    public static void main(String[] args) throws Exception {
        Job job = Job.getInstance(new Configuration(), "Log Count");
        job.setJarByClass(Driver.class);
        job.setMapperClass(Mapper1.class);
        job.setReducerClass(Reducer1.class);
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(IntWritable.class);
        FileInputFormat.addInputPath(job, new Path("/input"));
        FileOutputFormat.setOutputPath(job, new Path("/output"));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}
